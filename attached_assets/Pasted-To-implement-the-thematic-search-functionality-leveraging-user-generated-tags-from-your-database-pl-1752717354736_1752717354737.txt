async searchUnified(query: string, options: SearchOptions = {}): Promise<UnifiedSearchResults> {
  if (query.length < 2) {
    throw new Error('Search query must be at least 2 characters');
  }
  let searchUrl = `/api/search/unified?q=${encodeURIComponent(query)}`;
  // Add tags if provided
  if (options.tags) {
    searchUrl += `&tags=${encodeURIComponent(options.tags.join(','))}`;
  }
  // Existing location parameters...
  const response = await fetch(searchUrl, {
    method: 'GET',
    headers: { 'Content-Type': 'application/json' },
    signal: AbortSignal.timeout(8000),
  });
  // Handle response...
  const results = await response.json();
  // Sort by relevance considering both title and tags
  if (results.restaurants) {
    results.restaurants.sort((a: SearchResult, b: SearchResult) => {
      const aTags = a.tags || [];
      const bTags = b.tags || [];
      const aTagMatch = aTags.some(tag => tag.toLowerCase().includes(query.toLowerCase()));
      const bTagMatch = bTags.some(tag => tag.toLowerCase().includes(query.toLowerCase()));
      // Prioritize tag match over names, then by ratings
      return (bTagMatch ? 1 : 0) - (aTagMatch ? 1 : 0) || (b.avgRating || 0) - (a.avgRating || 0);
    });
  }
  return results;
}
