import { useEffect, useState } from "react";
import { useParams } from "react-router-dom"; // Or your route lib
import { ListCard } from "@/components/ListCard";
import { CircleMetadata } from "./CircleMetadata";

export function CircleFeed() {
  const { circleId } = useParams();
  const [circle, setCircle] = useState(null);
  const [lists, setLists] = useState([]);

  useEffect(() => {
    const fetchFeed = async () => {
      const res1 = await fetch(`/api/circles/${circleId}`);
      const res2 = await fetch(`/api/circles/${circleId}/feed`);
      const c = await res1.json();
      const l = await res2.json();
      setCircle(c);
      setLists(l);
    };
    fetchFeed();
  }, [circleId]);

  if (!circle) return <p>Loading...</p>;

  return (
    <div className="p-4 space-y-6">
      <CircleMetadata {...circle} />

      {lists.length === 0 ? (
        <p className="text-muted-foreground">No lists have been shared yet.</p>
      ) : (
        <div className="space-y-4">
          {lists.map((list) => (
            <ListCard key={list.id} {...list} />
          ))}
        </div>
      )}
    </div>
  );
}

import { useEffect, useState } from "react";
import { useParams } from "react-router-dom";
import { AccessDeniedView } from "./AccessDeniedView";

export function PermissionGuard({ children }: { children: React.ReactNode }) {
  const { circleId } = useParams();
  const [allowed, setAllowed] = useState<boolean | null>(null);

  useEffect(() => {
    const checkAccess = async () => {
      const res = await fetch(`/api/circles/${circleId}/access`);
      const data = await res.json();
      setAllowed(data.allowed);
    };
    checkAccess();
  }, [circleId]);

  if (allowed === null) return <p>Loading...</p>;
  if (!allowed) return <AccessDeniedView />;

  return <>{children}</>;
}
