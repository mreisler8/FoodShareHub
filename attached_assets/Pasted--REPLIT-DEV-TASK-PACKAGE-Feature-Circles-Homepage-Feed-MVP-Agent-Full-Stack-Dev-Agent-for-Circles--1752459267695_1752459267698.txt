export function TabNav({ currentTab, setTab }) {
  return (
    <div className="flex justify-around border-b text-sm font-medium">
      {["foryou", "trending", "nearby"].map((tab) => (
        <button
          key={tab}
          className={`py-2 px-4 ${
            currentTab === tab ? "border-b-2 border-black" : "text-muted-foreground"
          }`}
          onClick={() => setTab(tab)}
        >
          {tab === "foryou" && "For You"}
          {tab === "trending" && "Trending Now"}
          {tab === "nearby" && "Near You"}
        </button>
      ))}
    </div>
  );
}
