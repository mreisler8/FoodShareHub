import { useNavigate, useLocation } from "react-router-dom";
import { routes } from "@/lib/routes";

export default function BackButton({ fallback = routes.feed }: { fallback?: string }) {
  const nav = useNavigate();
  const { key } = useLocation(); // 'default' means no prior history in this tab
  const canGoBack = key !== "default";

  return (
    <button
      type="button"
      aria-label="Go back"
      onClick={() => (canGoBack ? nav(-1) : nav(fallback))}
      className="inline-flex items-center gap-2 rounded-2xl px-3 py-2 text-sm hover:bg-gray-100"
    >
      <span aria-hidden>←</span>
      <span>Back</span>
    </button>
  );
}

import { Link, useLocation } from "react-router-dom";
import { routes } from "@/lib/routes";
import BackButton from "@/components/ui/BackButton";

export default function AppHeader() {
  const { pathname } = useLocation();
  const showBack = pathname !== routes.feed;

  return (
    <header className="sticky top-0 z-40 w-full border-b bg-white/80 backdrop-blur supports-[backdrop-filter]:bg-white/60">
      <div className="mx-auto flex h-14 max-w-screen-md items-center justify-between px-3">
        <div className="flex items-center gap-2">
          {showBack ? <BackButton /> : null}
          <Link to={routes.feed} className="flex items-center gap-2" aria-label="Go to feed">
            {/* Temporarily render a text mark; swap in your logo later */}

            <span className="font-semibold">Circles</span>
          </Link>

      </div>
    </header>
  );
}
