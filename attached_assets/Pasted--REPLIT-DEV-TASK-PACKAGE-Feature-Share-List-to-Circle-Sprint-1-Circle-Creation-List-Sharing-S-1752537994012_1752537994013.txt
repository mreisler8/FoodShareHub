import { useEffect, useState } from "react";

interface Circle {
  id: string;
  name: string;
}

interface Props {
  value: { visibility: string; circleId?: string };
  onChange: (val: { visibility: string; circleId?: string }) => void;
}

export function ShareToDestinationDropdown({ value, onChange }: Props) {
  const [circles, setCircles] = useState<Circle[]>([]);

  useEffect(() => {
    const fetchCircles = async () => {
      const res = await fetch("/api/circles/mine");
      const data = await res.json();
      setCircles(data);
    };
    fetchCircles();
  }, []);

  return (
    <div className="space-y-2">
      <label className="block text-sm font-medium">Share this list to:</label>

      <select
        className="border px-2 py-1 rounded w-full"
        value={value.circleId || value.visibility}
        onChange={(e) => {
          const val = e.target.value;
          if (val === "profile" || val === "public") {
            onChange({ visibility: val });
          } else {
            onChange({ visibility: "circle", circleId: val });
          }
        }}
      >
        <option value="profile">Your Profile</option>
        <option value="public">Public Feed</option>
        {circles.map((c) => (
          <option key={c.id} value={c.id}>
            Circle: {c.name}
          </option>
        ))}
      </select>
    </div>
  );
}
