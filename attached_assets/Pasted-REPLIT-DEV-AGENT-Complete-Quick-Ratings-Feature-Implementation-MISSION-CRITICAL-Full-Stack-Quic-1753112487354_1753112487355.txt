import { Router } from 'express';
import { z } from 'zod';
import { db } from '../db';
import { ratings, insertRatingSchema, restaurants, circles, circleMembers } from '../../shared/schema';
import { eq, and, desc, asc, inArray } from 'drizzle-orm';
const router = Router();
// Validation schemas
const createRatingSchema = insertRatingSchema.extend({
  ratingValue: z.number().min(1).max(5),
  note: z.string().max(140).optional(),
  tags: z.array(z.string()).max(5).optional(),
});
const updateRatingSchema = createRatingSchema.partial().extend({
  id: z.number()
});
// GET /api/ratings - Get user's ratings with filters
router.get('/', async (req, res) => {
  if (!req.user?.id) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const { restaurantId, googlePlaceId, sharedOnly, limit = 50, offset = 0 } = req.query;
    
    let query = db.select().from(ratings).where(eq(ratings.userId, req.user.id));
    
    if (restaurantId) {
      query = query.where(eq(ratings.restaurantId, parseInt(restaurantId as string)));
    }
    
    if (googlePlaceId) {
      query = query.where(eq(ratings.googlePlaceId, googlePlaceId as string));
    }
    
    if (sharedOnly === 'true') {
      query = query.where(eq(ratings.sharedWithCircle, true));
    }
    
    query = query
      .orderBy(desc(ratings.createdAt))
      .limit(parseInt(limit as string))
      .offset(parseInt(offset as string));
    const userRatings = await query;
    res.json(userRatings);
  } catch (error) {
    console.error('Get ratings error:', error);
    res.status(500).json({ error: 'Failed to fetch ratings' });
  }
});
// GET /api/ratings/restaurant/:id - Get user's rating for specific restaurant
router.get('/restaurant/:id', async (req, res) => {
  if (!req.user?.id) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const { id } = req.params;
    const { type = 'restaurant' } = req.query; // 'restaurant' or 'google_place'
    
    let rating;
    if (type === 'google_place') {
      rating = await db.select().from(ratings)
        .where(and(
          eq(ratings.userId, req.user.id),
          eq(ratings.googlePlaceId, id)
        ))
        .limit(1);
    } else {
      rating = await db.select().from(ratings)
        .where(and(
          eq(ratings.userId, req.user.id),
          eq(ratings.restaurantId, parseInt(id))
        ))
        .limit(1);
    }
    if (rating.length === 0) {
      return res.status(404).json({ error: 'Rating not found' });
    }
    res.json(rating[0]);
  } catch (error) {
    console.error('Get restaurant rating error:', error);
    res.status(500).json({ error: 'Failed to fetch rating' });
  }
});
// POST /api/ratings - Create or update rating
router.post('/', async (req, res) => {
  if (!req.user?.id) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const validatedData = createRatingSchema.parse(req.body);
    
    // Validate circle access if sharing with circles
    if (validatedData.sharedWithCircle && validatedData.circleIds?.length) {
      const userCircles = await db.select()
        .from(circleMembers)
        .where(eq(circleMembers.userId, req.user.id));
      
      const userCircleIds = userCircles.map(cm => cm.circleId);
      const invalidCircles = validatedData.circleIds.filter(cId => !userCircleIds.includes(cId));
      
      if (invalidCircles.length > 0) {
        return res.status(403).json({ 
          error: 'Cannot share to circles you are not a member of' 
        });
      }
    }
    const ratingData = {
      ...validatedData,
      userId: req.user.id,
      updatedAt: new Date(),
    };
    // Check if rating already exists (upsert logic)
    let existingRating;
    if (validatedData.restaurantId) {
      existingRating = await db.select().from(ratings)
        .where(and(
          eq(ratings.userId, req.user.id),
          eq(ratings.restaurantId, validatedData.restaurantId)
        ))
        .limit(1);
    } else if (validatedData.googlePlaceId) {
      existingRating = await db.select().from(ratings)
        .where(and(
          eq(ratings.userId, req.user.id),
          eq(ratings.googlePlaceId, validatedData.googlePlaceId)
        ))
        .limit(1);
    }
    let rating;
    if (existingRating && existingRating.length > 0) {
      // Update existing rating
      rating = await db.update(ratings)
        .set(ratingData)
        .where(eq(ratings.id, existingRating[0].id))
        .returning();
    } else {
      // Create new rating
      rating = await db.insert(ratings)
        .values(ratingData)
        .returning();
    }
    res.status(201).json(rating[0]);
  } catch (error) {
    if (error instanceof z.ZodError) {
      return res.status(400).json({ 
        error: 'Validation error', 
        details: error.errors 
      });
    }
    
    console.error('Create rating error:', error);
    res.status(500).json({ error: 'Failed to create rating' });
  }
});
// PUT /api/ratings/:id - Update existing rating
router.put('/:id', async (req, res) => {
  if (!req.user?.id) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const ratingId = parseInt(req.params.id);
    const validatedData = updateRatingSchema.parse({ ...req.body, id: ratingId });
    // Verify ownership
    const existingRating = await db.select().from(ratings)
      .where(and(
        eq(ratings.id, ratingId),
        eq(ratings.userId, req.user.id)
      ))
      .limit(1);
    if (existingRating.length === 0) {
      return res.status(404).json({ error: 'Rating not found or access denied' });
    }
    // Validate circle access if updating circle sharing
    if (validatedData.sharedWithCircle && validatedData.circleIds?.length) {
      const userCircles = await db.select()
        .from(circleMembers)
        .where(eq(circleMembers.userId, req.user.id));
      
      const userCircleIds = userCircles.map(cm => cm.circleId);
      const invalidCircles = validatedData.circleIds.filter(cId => !userCircleIds.includes(cId));
      
      if (invalidCircles.length > 0) {
        return res.status(403).json({ 
          error: 'Cannot share to circles you are not a member of' 
        });
      }
    }
    const updateData = {
      ...validatedData,
      updatedAt: new Date(),
    };
    delete updateData.id; // Remove id from update data
    const updatedRating = await db.update(ratings)
      .set(updateData)
      .where(eq(ratings.id, ratingId))
      .returning();
    res.json(updatedRating[0]);
  } catch (error) {
    if (error instanceof z.ZodError) {
      return res.status(400).json({ 
        error: 'Validation error', 
        details: error.errors 
      });
    }
    
    console.error('Update rating error:', error);
    res.status(500).json({ error: 'Failed to update rating' });
  }
});
// DELETE /api/ratings/:id - Delete rating
router.delete('/:id', async (req, res) => {
  if (!req.user?.id) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const ratingId = parseInt(req.params.id);
    // Verify ownership
    const existingRating = await db.select().from(ratings)
      .where(and(
        eq(ratings.id, ratingId),
        eq(ratings.userId, req.user.id)
      ))
      .limit(1);
    if (existingRating.length === 0) {
      return res.status(404).json({ error: 'Rating not found or access denied' });
    }
    await db.delete(ratings).where(eq(ratings.id, ratingId));
    
    res.json({ message: 'Rating deleted successfully' });
  } catch (error) {
    console.error('Delete rating error:', error);
    res.status(500).json({ error: 'Failed to delete rating' });
  }
});
// GET /api/ratings/circle/:circleId - Get ratings shared with specific circle
router.get('/circle/:circleId', async (req, res) => {
  if (!req.user?.id) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const circleId = parseInt(req.params.circleId);
    // Verify circle membership
    const membership = await db.select().from(circleMembers)
      .where(and(
        eq(circleMembers.circleId, circleId),
        eq(circleMembers.userId, req.user.id),
        eq(circleMembers.status, 'active')
      ))
      .limit(1);
    if (membership.length === 0) {
      return res.status(403).json({ error: 'Access denied to this circle' });
    }
    // Get ratings shared with this circle
    const circleRatings = await db.select()
      .from(ratings)
      .where(and(
        eq(ratings.sharedWithCircle, true),
        inArray(ratings.circleIds, [circleId])
      ))
      .orderBy(desc(ratings.createdAt));
    res.json(circleRatings);
  } catch (error) {
    console.error('Get circle ratings error:', error);
    res.status(500).json({ error: 'Failed to fetch circle ratings' });
  }
});
export default router;

import { Star, Heart, Coffee, Users, Clock, DollarSign, X } from 'lucide-react';

interface QuickRatingModalProps {

  onClose: () => void;

    name: string;

    priceRange?: string;

const QUICK_TAGS = [
  { value: 'great-value', label: 'Great Value', icon: DollarSign },
  { value: 'perfect-date-spot', label: 'Perfect Date Spot', icon: Heart },
  { value: 'quick-bite', label: 'Quick Bite', icon: Clock },
  { value: 'group-friendly', label: 'Group Friendly', icon: Users },
  { value: 'coffee-spot', label: 'Coffee Spot', icon: Coffee },
  { value: 'hidden-gem', label: 'Hidden Gem', icon: Star },
];
export function QuickRatingModal({ isOpen, onClose, restaurant, existingRating }: QuickRatingModalProps) {
  const [ratingValue, setRatingValue] = useState(existingRating?.ratingValue || 0);
  const [note, setNote] = useState(existingRating?.note || '');
  const [selectedTags, setSelectedTags] = useState<string[]>(existingRating?.tags || []);
  const [sharedWithCircle, setSharedWithCircle] = useState(existingRating?.sharedWithCircle || false);
  const [selectedCircles, setSelectedCircles] = useState<number[]>(existingRating?.circleIds || []);
  const queryClient = useQueryClient();
  // Fetch user's circles
  const { data: circles } = useQuery({
    queryKey: ['/api/circles'],
    queryFn: async () => {
      const response = await fetch('/api/circles');
      if (!response.ok) throw new Error('Failed to fetch circles');
      return response.json();
    },
  });
  const circleOptions = circles?.map((circle: any) => ({
    value: circle.id.toString(),
    label: circle.name,
  })) || [];
  // Create/Update rating mutation
  const createRatingMutation = useMutation({
    mutationFn: async (data: any) => {
      const url = existingRating ? `/api/ratings/${existingRating.id}` : '/api/ratings';
      const method = existingRating ? 'PUT' : 'POST';
      
      const response = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data),
      });
      
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error || 'Failed to save rating');
      }
      
      return response.json();
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['/api/ratings'] });
      queryClient.invalidateQueries({ queryKey: [`/api/ratings/restaurant/${restaurant.id || restaurant.googlePlaceId}`] });
      onClose();
    },
  });
  const handleSubmit = async () => {
    if (ratingValue === 0) return;
    const ratingData = {
      restaurantId: restaurant.id || null,
      googlePlaceId: restaurant.googlePlaceId || null,
      restaurantName: restaurant.name,
      ratingValue,
      note: note.trim() || null,
      tags: selectedTags,
      sharedWithCircle,
      circleIds: sharedWithCircle ? selectedCircles : [],
      isPrivate: !sharedWithCircle,
    };
    createRatingMutation.mutate(ratingData);
  };
  const handleTagToggle = (tagValue: string) => {
    setSelectedTags(prev => 
      prev.includes(tagValue) 
        ? prev.filter(t => t !== tagValue)
        : [...prev, tagValue].slice(0, 3) // Max 3 tags
    );
  };
  return (
    <Dialog open={isOpen} onOpenChange={onClose}>
      <DialogContent className="sm:max-w-[500px] max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2">
            <Star className="h-5 w-5 text-yellow-500" />

          </DialogTitle>
        </DialogHeader>
        <div className="space-y-6">
          {/* Restaurant Info */}
          <div className="bg-gray-50 rounded-lg p-3">
            <h3 className="font-medium">{restaurant.name}</h3>
            {restaurant.cuisine && (
              <p className="text-sm text-gray-600">{restaurant.cuisine}</p>
            )}
            {restaurant.address && (
              <p className="text-xs text-gray-500 mt-1">{restaurant.address}</p>
            )}
          </div>
          {/* Star Rating */}
          <div className="text-center">
            <Label className="block text-sm font-medium mb-3">Your Rating</Label>
            <div className="flex justify-center">
              <Rating
                value={ratingValue}
                onChange={setRatingValue}
                size="lg"
                readonly={false}
                showValue={false}
              />

              <p className="text-sm text-gray-600 mt-2">
                {ratingValue === 5 ? 'Amazing!' :

                 ratingValue === 3 ? 'Good' : 
                 ratingValue === 2 ? 'Okay' : 'Not great'}
              </p>
            )}
          </div>
          {/* Quick Tags */}
          <div>
            <Label className="block text-sm font-medium mb-3">Quick Tags (up to 3)</Label>
            <div className="grid grid-cols-2 gap-2">
              {QUICK_TAGS.map((tag) => {
                const Icon = tag.icon;
                const isSelected = selectedTags.includes(tag.value);
                
                return (
                  <button
                    key={tag.value}
                    onClick={() => handleTagToggle(tag.value)}
                    disabled={!isSelected && selectedTags.length >= 3}
                    className={cn(
                      "flex items-center gap-2 p-2 rounded-lg border text-sm transition-colors",
                      isSelected
                        ? "bg-primary/10 border-primary text-primary"
                        : "bg-white border-gray-200 text-gray-700 hover:bg-gray-50",

import React, { useState } from 'react';
import { Button } from "@/components/ui/button";
import { Star } from 'lucide-react';
import { QuickRatingModal } from './QuickRatingModal';
import { useQuery } from '@tanstack/react-query';
import { cn } from "@/lib/utils";
interface RatingButtonProps {
  restaurant: {
    id?: number;
    googlePlaceId?: string;
    name: string;
    cuisine?: string;
    priceRange?: string;
    address?: string;
  };
  variant?: 'default' | 'outline' | 'ghost';
  size?: 'sm' | 'md' | 'lg';
  showRating?: boolean;
  className?: string;
}
export function RatingButton({ 
  restaurant, 
  variant = 'outline', 
  size = 'sm',
  showRating = true,
  className 
}: RatingButtonProps) {
  const [isModalOpen, setIsModalOpen] = useState(false);
  // Fetch existing rating
  const { data: existingRating, isLoading } = useQuery({
    queryKey: [`/api/ratings/restaurant/${restaurant.id || restaurant.googlePlaceId}`],
    queryFn: async () => {
      const id = restaurant.id || restaurant.googlePlaceId;
      const type = restaurant.id ? 'restaurant' : 'google_place';
      
      const response = await fetch(`/api/ratings/restaurant/${id}?type=${type}`);
      if (response.status === 404) return null;
      if (!response.ok) throw new Error('Failed to fetch rating');
      return response.json();
    },
    enabled: !!(restaurant.id || restaurant.googlePlaceId),
  });
  const hasRating = !!existingRating;
  const ratingValue = existingRating?.ratingValue || 0;
  const getSizeStyles = () => {
    switch (size) {
      case 'sm': return 'h-8 px-3 text-xs';
      case 'lg': return 'h-12 px-6 text-base';
      default: return 'h-10 px-4 text-sm';
    }
  };
  const getIconSize = () => {
    switch (size) {
      case 'sm': return 'h-3 w-3';
      case 'lg': return 'h-5 w-5';
      default: return 'h-4 w-4';
    }
  };
  return (
    <>
      <Button
        variant={hasRating ? 'default' : variant}
        onClick={() => setIsModalOpen(true)}
        disabled={isLoading}
        className={cn(
          getSizeStyles(),
          hasRating && 'bg-yellow-500 hover:bg-yellow-600 text-white',
          className
        )}
      >
        <Star 
          className={cn(
            getIconSize(),
            'mr-1.5',
            hasRating && 'fill-current'
          )} 
        />
        {hasRating ? (
          showRating ? (
            <span>{ratingValue}/5</span>
          ) : (
            <span>Rated</span>
          )
        ) : (
          <span>Rate</span>
        )}
      </Button>
      <QuickRatingModal
        isOpen={isModalOpen}
        onClose={() => setIsModalOpen(false)}
        restaurant={restaurant}
        existingRating={existingRating}
      />
    </>
  );
}

  };
  showNote?: boolean;
  showTags?: boolean;
  showTimestamp?: boolean;
  compact?: boolean;
  className?: string;
}
const TAG_LABELS: Record<string, string> = {
  'great-value': 'Great Value',
  'perfect-date-spot': 'Perfect Date Spot',
  'quick-bite': 'Quick Bite',
  'group-friendly': 'Group Friendly',
  'coffee-spot': 'Coffee Spot',
  'hidden-gem': 'Hidden Gem',
};
export function RatingDisplay({ 
  rating, 
  showNote = true, 
  showTags = true, 
  showTimestamp = false,
  compact = false,
  className 
}: RatingDisplayProps) {
  const formatDate = (dateString: string) => {
    const date = new Date(dateString);
    const now = new Date();
    const diffMs = now.getTime() - date.getTime();
    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
    
    if (diffDays === 0) return 'Today';
    if (diffDays === 1) return 'Yesterday';
    if (diffDays < 7) return `${diffDays} days ago`;
    if (diffDays < 30) return `${Math.floor(diffDays / 7)} weeks ago`;
    return date.toLocaleDateString();
  };
  if (compact) {
    return (
      <div className={cn("flex items-center gap-2", className)}>
        <div className="flex items-center gap-1">
          <Star className="h-4 w-4 text-yellow-500 fill-current" />
          <span className="text-sm font-medium">{rating.ratingValue}/5</span>
        </div>
        {rating.sharedWithCircle && (
          <Badge variant="secondary" className="text-xs">
            Shared
          </Badge>
        )}
      </div>
    );
  }
  return (
    <div className={cn("space-y-3", className)}>
      {/* Rating Value */}
      <div className="flex items-center justify-between">
        <Rating 
          value={rating.ratingValue} 
          size="md" 
          showValue={true}
          readonly={true}
        />
        {rating.sharedWithCircle && (
          <Badge variant="secondary">
            Shared with Circles
          </Badge>
        )}
      </div>
      {/* Tags */}
      {showTags && rating.tags && rating.tags.length > 0 && (
        <div className="flex flex-wrap gap-2">
          {rating.tags.map((tag, index) => (
            <Badge key={index} variant="outline" className="text-xs">
              {TAG_LABELS[tag] || tag}
            </Badge>
          ))}
        </div>
      )}
      {/* Note */}
      {showNote && rating.note && (
        <div className="bg-gray-50 rounded-lg p-3">
          <p className="text-sm text-gray-700">{rating.note}</p>
        </div>
      )}
      {/* Timestamp */}
      {showTimestamp && (
        <div className="flex items-center gap-1 text-xs text-gray-500">
          <Clock className="h-3 w-3" />
          <span>Rated {formatDate(rating.createdAt)}</span>
        </div>
      )}
    </div>
  );
}

import React, { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { RatingDisplay } from './RatingDisplay';
import { QuickRatingModal } from './QuickRatingModal';
import { Star, Edit, Filter, SortDesc } from 'lucide-react';
import { cn } from "@/lib/utils";
interface MyRatingsListProps {
  limit?: number;
  showFilters?: boolean;
  className?: string;
}
export function MyRatingsList({ 
  limit = 20, 
  showFilters = false,
  className 
}: MyRatingsListProps) {
  const [editingRating, setEditingRating] = useState<any>(null);
  const [filter, setFilter] = useState<'all' | 'shared' | 'private'>('all');
  // Fetch user's ratings
  const { data: ratings, isLoading, error } = useQuery({
    queryKey: ['/api/ratings', { limit, sharedOnly: filter === 'shared' }],
    queryFn: async () => {
      let url = `/api/ratings?limit=${limit}`;
      if (filter === 'shared') url += '&sharedOnly=true';
      
      const response = await fetch(url);
      if (!response.ok) throw new Error('Failed to fetch ratings');
      return response.json();
    },
  });
  if (isLoading) {
    return (
      <Card className={className}>
        <CardContent className="p-6">
          <div className="animate-pulse space-y-4">
            {[...Array(3)].map((_, i) => (
              <div key={i} className="h-24 bg-gray-200 rounded-lg" />
            ))}
          </div>
        </CardContent>
      </Card>
    );
  }
  if (error) {
    return (
      <Card className={className}>
        <CardContent className="p-6 text-center text-gray-500">
          Failed to load your ratings
        </CardContent>
      </Card>
    );
  }
  const filteredRatings = ratings?.filter((rating: any) => {
    if (filter === 'all') return true;
    if (filter === 'shared') return rating.sharedWithCircle;
    if (filter === 'private') return !rating.sharedWithCircle;
    return true;
  }) || [];
  return (
    <>
      <Card className={className}>
        <CardHeader className="pb-4">
          <div className="flex items-center justify-between">
            <CardTitle className="flex items-center gap-2">
              <Star className="h-5 w-5 text-yellow-500" />
              My Ratings
            </CardTitle>
            {showFilters && (
              <div className="flex items-center gap-2">
                <Button
                  variant={filter === 'all' ? 'default' : 'ghost'}
                  size="sm"
                  onClick={() => setFilter('all')}
                >
                  All
                </Button>
                <Button
                  variant={filter === 'shared' ? 'default' : 'ghost'}
                  size="sm"
                  onClick={() => setFilter('shared')}
                >
                  Shared
                </Button>
                <Button
                  variant={filter === 'private' ? 'default' : 'ghost'}
                  size="sm"
                  onClick={() => setFilter('private')}
                >
                  Private
                </Button>
              </div>
            )}
          </div>
        </CardHeader>
        
        <CardContent>
          {filteredRatings.length === 0 ? (
            <div className="text-center py-8">
              <Star className="h-12 w-12 text-gray-300 mx-auto mb-4" />
              <p className="text-gray-500 mb-2">No ratings yet</p>
              <p className="text-sm text-gray-400">
                Start rating restaurants to see them here
              </p>
            </div>
          ) : (
            <div className="space-y-4">
              {filteredRatings.map((rating: any) => (
                <div 
                  key={rating.id} 
                  className="bg-gray-50 rounded-lg p-4 space-y-3"
                >
                  {/* Restaurant Info */}
                  <div className="flex items-start justify-between">
                    <div className="flex-1 min-w-0">
                      <h3 className="font-medium truncate">
                        {rating.restaurantName}
                      </h3>
                    </div>
                    <Button
                      variant="ghost"
                      size="sm"
                      onClick={() => setEditingRating(rating)}
                      className="ml-2 shrink-0"
                    >
                      <Edit className="h-4 w-4" />
                    </Button>
                  </div>
                  {/* Rating Display */}
                  <RatingDisplay 
                    rating={rating}
                    showTimestamp={true}
                    compact={false}
                  />
                </div>
              ))}
              
              {filteredRatings.length === limit && (
                <div className="text-center pt-4">
                  <Button variant="outline" size="sm">
                    Load More
                  </Button>
                </div>
              )}
            </div>
          )}
        </CardContent>
      </Card>
      {/* Edit Rating Modal */}
      {editingRating && (
        <QuickRatingModal
          isOpen={!!editingRating}
          onClose={() => setEditingRating(null)}
          restaurant={{
            id: editingRating.restaurantId,
            googlePlaceId: editingRating.googlePlaceId,
            name: editingRating.restaurantName,
          }}
          existingRating={editingRating}
        />
      )}
    </>
  );
}
