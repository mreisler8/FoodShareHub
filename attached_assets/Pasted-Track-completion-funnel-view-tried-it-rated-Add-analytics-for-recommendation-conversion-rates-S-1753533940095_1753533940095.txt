CORE REQUIREMENTS:

1. DATABASE SCHEMA

CREATE TABLE accepted_recommendations (
  id SERIAL PRIMARY KEY,
  actor_user_id INTEGER NOT NULL REFERENCES users(id),
  recommender_user_id INTEGER NOT NULL REFERENCES users(id),
  entity_type VARCHAR(20) NOT NULL CHECK (entity_type IN ('list', 'rating', 'post')),
  entity_id INTEGER NOT NULL,
  restaurant_id INTEGER NOT NULL,
  source_context VARCHAR(50), -- 'list_item', 'restaurant_page', 'rating_card', etc.
  rating_value INTEGER CHECK (rating_value BETWEEN 1 AND 5),
  notes TEXT,
  created_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(actor_user_id, entity_type, entity_id, restaurant_id)
);
2. API ENDPOINTS

POST /api/recommendations/accept - Record acceptance with duplicate prevention
GET /api/recommendations/accepted/:userId - User's accepted recommendations
GET /api/recommendations/impact/:userId - Recommender's influence stats
GET /api/recommendations/stats/:entityType/:entityId - Get acceptance count for content
